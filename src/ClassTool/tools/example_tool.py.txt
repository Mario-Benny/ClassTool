# -*- coding: utf-8 -*-

"""
ClassTool example plugin.

How to use:
1. Copy this file into the tools/ directory.
2. Rename it, for example: my_first_tool.py.
3. Edit TOOL_INFO and the ExampleToolWidget UI.
4. Restart ClassTool, or wait for the tools directory watcher to refresh.

Plugin rules:
- TOOL_INFO must be a literal dictionary because ClassTool reads it with ast.literal_eval.
- Provide create_widget(), get_widget(), or run().
- create_widget() should return a QWidget instance.
"""

from PySide6.QtCore import Qt
from PySide6.QtWidgets import (
    QWidget,
    QVBoxLayout,
    QHBoxLayout,
    QLabel,
    QPushButton,
    QLineEdit,
    QMessageBox,
)

from toolkit.core.tools_config import get_tool_config, set_tool_config


TOOL_INFO = {
    "name": "示例插件",
    "color": "#2D89EF",
    "icon": "\uECAA",
    "description": "第三方插件开发模板",
    "requirements": ["PySide6.QtCore", "PySide6.QtWidgets"],
    "cache": True,
}


CONFIG_KEY = "example_tool"


def _load_config() -> dict:
    config = get_tool_config(CONFIG_KEY)
    return config if isinstance(config, dict) else {}


def _save_config(data: dict):
    set_tool_config(CONFIG_KEY, data)


class ExampleToolWidget(QWidget):
    """A small QWidget-based tool that demonstrates UI and config storage."""

    def __init__(self):
        super().__init__()
        self._setup_ui()
        self._load_state()

    def _setup_ui(self):
        self.setStyleSheet("background-color: #1e1e1e;")

        root = QVBoxLayout(self)
        root.setContentsMargins(30, 30, 30, 30)
        root.setSpacing(16)

        title = QLabel("示例插件")
        title.setStyleSheet("color: white; font-size: 22pt; font-weight: bold;")
        root.addWidget(title)

        intro = QLabel(
            "这是一个最小但完整的 ClassTool 插件模板。"
            "你可以在这里替换自己的界面、逻辑和配置项。"
        )
        intro.setWordWrap(True)
        intro.setStyleSheet("color: #cccccc; font-size: 11pt;")
        root.addWidget(intro)

        row = QHBoxLayout()
        label = QLabel("课堂提示：")
        label.setStyleSheet("color: #dddddd; font-size: 11pt;")
        row.addWidget(label)

        self.message_input = QLineEdit()
        self.message_input.setPlaceholderText("输入一条会被保存的提示")
        self.message_input.setStyleSheet("""
            QLineEdit {
                background-color: #2b2b2b;
                color: white;
                border: 1px solid #444444;
                padding: 8px 10px;
                font-size: 11pt;
            }
            QLineEdit:focus {
                border-color: #2D89EF;
            }
        """)
        row.addWidget(self.message_input, 1)
        root.addLayout(row)

        self.preview_label = QLabel("")
        self.preview_label.setAlignment(Qt.AlignmentFlag.AlignCenter)
        self.preview_label.setWordWrap(True)
        self.preview_label.setStyleSheet("""
            QLabel {
                background-color: #111111;
                color: #4FC3F7;
                border: 1px solid #333333;
                padding: 20px;
                font-size: 18pt;
                font-weight: bold;
            }
        """)
        root.addWidget(self.preview_label, 1)

        button_row = QHBoxLayout()

        save_btn = QPushButton("保存并预览")
        save_btn.clicked.connect(self.save_message)
        save_btn.setStyleSheet(self._button_style("#0078D7", "#1E88E5"))
        button_row.addWidget(save_btn)

        clear_btn = QPushButton("清空")
        clear_btn.clicked.connect(self.clear_message)
        clear_btn.setStyleSheet(self._button_style("#3a3a3a", "#4a4a4a"))
        button_row.addWidget(clear_btn)

        root.addLayout(button_row)

    @staticmethod
    def _button_style(color: str, hover_color: str) -> str:
        return f"""
            QPushButton {{
                background-color: {color};
                color: white;
                border: none;
                border-radius: 6px;
                padding: 10px 16px;
                font-size: 11pt;
                font-weight: bold;
            }}
            QPushButton:hover {{ background-color: {hover_color}; }}
            QPushButton:pressed {{ background-color: #1f1f1f; }}
        """

    def _load_state(self):
        message = _load_config().get("message", "")
        if not isinstance(message, str):
            message = ""
        self.message_input.setText(message)
        self._set_preview(message)

    def _set_preview(self, message: str):
        text = message.strip() or "这里会显示插件保存的内容"
        self.preview_label.setText(text)

    def save_message(self):
        message = self.message_input.text().strip()
        _save_config({"message": message})
        self._set_preview(message)
        QMessageBox.information(self, "保存成功", "示例插件配置已保存。")

    def clear_message(self):
        self.message_input.clear()
        _save_config({"message": ""})
        self._set_preview("")


def create_widget():
    """ClassTool calls this function when the user opens the tool."""
    return ExampleToolWidget()
